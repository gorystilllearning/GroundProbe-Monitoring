  SYSTEM REQUIREMENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Operating System  : Windows 10/11  |  macOS 12+  |  Ubuntu 22.04
  RAM               : Minimum 4 GB  (8 GB recommended)
  Storage           : Minimum 20 GB free space
  Docker Desktop    : Required — download at docker.com/get-started

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  INSTALLATION — WINDOWS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Step 1 : Install wsl_update before download docker

  Step 2 : Install Docker Desktop from docker.com/get-started
           Restart your PC after installation

  Step 3 : Extract this zip file to any folder

  Step 4 : Make sure Docker Desktop is running
           (check system tray at bottom right)

  Step 5 : Double click "setup.bat"
           Wait approximately 90 seconds

  Step 6 : Open your browser
           Go to: http://localhost:8080

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  INSTALLATION — macOS / LINUX
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Step 1 : Install Docker Desktop from docker.com/get-started
           Restart your Mac after installation

  Step 2 : Extract this zip file to any folder

  Step 3 : Open Terminal and navigate to the extracted folder:
           cd path/to/groundprobe-monitoring

  Step 4 : Run the setup script:
           ./setup.sh
           Wait approximately 90 seconds

  Step 5 : Open your browser
           Go to: http://localhost:8080
